# 像素猫精灵 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把程序化矢量橘猫替换为 96×96 像素精灵（🐈 emoji 打底、橘白 4 色），用「单精灵 + 运行时变换」实现各状态动画。

**Architecture:** 用 Python（Playwright + Pillow）一次性生成 `cat_sprite.h`（RGB565 精灵 + 1-bit 透明遮罩）提交进仓库；固件 `pet_animation.cpp` 的 `draw()` 从「程序化画形状」改为「精灵 blit + 镜像/bob/squash/灰化/变暗」；`update()` 状态机不动，符号（`?`/`Zzz`/⭐/💧）与蝴蝶保留。

**Tech Stack:** PlatformIO + Arduino-ESP32、Adafruit GFX、Python（Playwright + Pillow，仅资产生成用）。

```python
#!/usr/bin/env python3
"""Generate the 96x96 orange-white cat sprite header from the 🐈 emoji."""
from playwright.sync_api import sync_playwright
from PIL import Image
import io
from pathlib import Path

OUT = Path(__file__).resolve().parent.parent / "firmware" / "src" / "ui" / "cat_sprite.h"
# Playwright chromium (headless shell) already installed; path may vary by version.
CHROMIUM = None
import glob, os
for p in sorted(glob.glob(os.path.expanduser("~/Library/Caches/ms-playwright/chromium_headless_shell-*/chrome-headless-shell-mac-x64/chrome-headless-shell")), reverse=True):
    CHROMIUM = p
    break
if not CHROMIUM:
    raise SystemExit("chromium headless shell not found; run: playwright install")

# orange + white (no pink)
PAL = [(0x40, 0x2a, 0x1c), (0xf6, 0xc8, 0x80), (0xd6, 0x9c, 0x58), (0xfc, 0xe2, 0xb4)]
SIZE = 96

def render(size=1024):
    html = f'''<html><head><meta charset="utf-8"></head>
    <body style="margin:0;background:transparent;display:flex;align-items:center;justify-content:center;width:{size}px;height:{size}px;overflow:hidden;">
    <span style="font-family:'Apple Color Emoji','Segoe UI Emoji','Noto Color Emoji',sans-serif;font-size:{int(size*0.92)}px;line-height:1;">🐈</span></body></html>'''
    with sync_playwright() as p:
        b = p.chromium.launch(executable_path=CHROMIUM, args=["--force-color-profile=srgb"])
        pg = b.new_page(viewport={"width": size, "height": size})
        pg.set_content(html); pg.wait_for_timeout(350)
        buf = pg.screenshot(omit_background=True); b.close()
    return Image.open(io.BytesIO(buf)).convert("RGBA")

def nearest(c):
    r, g, b = c
    best, bestd = PAL[0], 1e9
    for p in PAL:
        d = (r-p[0])**2 + (g-p[1])**2 + (b-p[2])**2
        if d < bestd: best, bestd = p, d
    return best

img = render().resize((SIZE, SIZE), Image.LANCZOS)
a = img.getchannel("A")
mask_bits = []
rgbs = []
for y in range(SIZE):
    for x in range(SIZE):
        opaque = a.getpixel((x, y)) > 48
        mask_bits.append(1 if opaque else 0)
        if opaque:
            r, g, b = nearest(img.getpixel((x, y))[:3])
            rgbs.append(((r & 0xF8) << 8) | ((g & 0xFC) << 3) | (b >> 3))
        else:
            rgbs.append(0)

# pack mask bits -> bytes (MSB first)
mask_bytes = []
for i in range(0, len(mask_bits), 8):
    byte = 0
    for j in range(8):
        byte = (byte << 1) | mask_bits[i + j]
    mask_bytes.append(byte)

lines = ["// Generated by scripts/gen_cat_sprite.py — do not edit by hand.",
         "#pragma once", "#include <cstdint>", "",
         f"// {SIZE}x{SIZE} orange-white cat sprite (RGB565) + 1-bit alpha mask.",
         f"constexpr int kCatSpriteW = {SIZE};",
         f"constexpr int kCatSpriteH = {SIZE};", "",
         f"constexpr uint16_t kCatSprite[{len(rgbs)}] = {{"]
for i in range(0, len(rgbs), 12):
    lines.append("    " + ", ".join(f"0x{v:04X}" for v in rgbs[i:i+12]) + ",")
lines += ["};", "", f"constexpr uint8_t kCatMask[{len(mask_bytes)}] = {{"]
for i in range(0, len(mask_bytes), 16):
    lines.append("    " + ", ".join(f"0x{b:02X}" for b in mask_bytes[i:i+16]) + ",")
lines += ["};", ""]

OUT.parent.mkdir(parents=True, exist_ok=True)
OUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
print(f"wrote {OUT} ({len(rgbs)} px, {len(mask_bytes)} mask bytes)")
```

```cpp
void PetAnimation::draw(Adafruit_ST7789& tft, int16_t x, int16_t y) {
    canvas_.fillScreen(COLOR_BG);

    const float t = static_cast<float>(millis() - stateStartedMs_) / 1000.0f;
    const int16_t cx = static_cast<int16_t>(walkX_);
    const int16_t jump = static_cast<int16_t>(bob_);
    const int16_t ground = 110 - jump;

    // ground shadow
    const bool offline = state_ == PetState::OFFLINE;
    const bool sleeping = state_ == PetState::SLEEP;
    canvas_.fillEllipse(cx, 112, 40, 5, offline ? rgb565(50, 50, 55) : rgb565(58, 58, 66));

    blitSprite(cx, ground);

    // butterfly: chase it while working, present it when completed
    if (state_ == PetState::WORKING) {
        drawButterfly2(static_cast<int16_t>(butterflyX_), static_cast<int16_t>(butterflyY_), sinf(t * 18.0f));
    } else if (state_ == PetState::COMPLETED) {
        drawButterfly2(cx + 24 * (facingLeft_ ? -1 : 1), ground - 40, 1.0f);
    }

    // symbols above the head
    drawSymbols2(cx, ground - kCatSpriteH - 8, t);

    const uint16_t* buf = canvas_.getBuffer();
    tft.startWrite();
    tft.setAddrWindow(x, y, CANVAS_W, CANVAS_H);
    uint32_t total = CANVAS_W * CANVAS_H;
    uint32_t off = 0;
    while (off < total) {
        const uint32_t n = (total - off > 1024) ? 1024 : (total - off);
        tft.writePixels(const_cast<uint16_t*>(buf + off), n);
        off += n;
    }
    tft.endWrite();
}
```

- [ ] **Step 3: 实现 blitSprite + 颜色变换**

在 `pet_animation.cpp` 追加：

```cpp
uint16_t PetAnimation::desaturate(uint16_t c) {
    uint32_t r = (c >> 11) & 0x1F, g = (c >> 5) & 0x3F, b = c & 0x1F;
    uint32_t lum = (r * 2 + g + b) / 4;          // approx luma (weight green)
    return (uint16_t)((lum << 11) | (lum << 5) | lum);
}

uint16_t PetAnimation::darken(uint16_t c) {
    return (uint16_t)((c >> 1) & 0x7BEF);         // ~50% brightness, keep RGB565
}

void PetAnimation::blitSprite(int16_t cx, int16_t groundY) {
    const bool offline = state_ == PetState::OFFLINE;
    const bool sleeping = state_ == PetState::SLEEP;
    const float sq = (state_ == PetState::SLEEP) ? 0.72f : 1.0f;  // slight vertical squash when sleeping
    const int H = static_cast<int>(kCatSpriteH * sq);
    const int topY = groundY - H;
    for (int dy = 0; dy < H; ++dy) {
        const int sy = static_cast<int>(dy * kCatSpriteH / (float)H);   // nearest source row
        for (int dx = 0; dx < kCatSpriteW; ++dx) {
            const int sx = facingLeft_ ? (kCatSpriteW - 1 - dx) : dx;   // mirror
            const int idx = sy * kCatSpriteW + sx;
            if (!(kCatMask[idx >> 3] & (0x80 >> (idx & 7)))) continue;  // transparent
            uint16_t c = kCatSprite[idx];
            if (offline) c = desaturate(c);
            else if (sleeping) c = darken(c);
            canvas_.drawPixel(cx - kCatSpriteW / 2 + dx, topY + dy, c);
        }
    }
}
```

```cpp
void PetAnimation::drawButterfly2(int16_t bx, int16_t by, float flap) {
    const int16_t wing = 6 + static_cast<int16_t>(flap * 2.0f);
    const uint16_t wingColor = rgb565(255, 190, 90);
    const uint16_t wingDark = rgb565(140, 100, 220);
    canvas_.fillTriangle(bx, by, bx - wing, by - 6, bx - 3, by + 2, wingColor);
    canvas_.fillTriangle(bx, by, bx + wing, by - 6, bx + 3, by + 2, wingColor);
    canvas_.fillTriangle(bx, by, bx - wing + 3, by + 6, bx - 2, by + 2, wingDark);
    canvas_.fillTriangle(bx, by, bx + wing - 3, by + 6, bx + 2, by + 2, wingDark);
    canvas_.fillRect(bx - 1, by - 4, 2, 8, COLOR_LINE);
    canvas_.drawLine(bx, by - 4, bx - 3, by - 8, COLOR_LINE);
    canvas_.drawLine(bx, by - 4, bx + 3, by - 8, COLOR_LINE);
}

void PetAnimation::drawSymbols2(int16_t cx, int16_t cy, float t) {
    canvas_.setTextSize(2);
    canvas_.setTextColor(COLOR_WHITE);
    if (showQuestion_) { canvas_.setCursor(cx - 8, cy); canvas_.print('?'); }
    else if (showDots_) { canvas_.setCursor(cx - 22, cy); canvas_.print("..."); }
    else if (showStar_) {
        const float tw = 0.5f + 0.5f * sinf(t * 6.0f);
        const uint16_t starColor = (tw > 0.5f) ? COLOR_STAR : rgb565(200, 160, 40);
        for (int i = -1; i <= 1; ++i) {
            const int16_t sx = cx + i * 22, sy = cy + 4 + ((i % 2) ? 10 : 0);
            canvas_.fillRect(sx - 2, sy - 6, 5, 12, starColor);
            canvas_.fillRect(sx - 6, sy - 2, 12, 5, starColor);
        }
    }
    else if (showSweat_) {
        canvas_.fillCircle(cx + 34, cy + 6, 4, COLOR_TEAL);
        canvas_.fillCircle(cx + 33, cy + 5, 2, COLOR_BG);
    }
    if (showZzz_) {
        canvas_.setTextColor(COLOR_TEAL);
        canvas_.setTextSize(1);
        const int off = static_cast<int>(t * 12.0f) % 24;
        canvas_.setCursor(cx + 18, cy + 14 - off); canvas_.print('z');
        canvas_.setCursor(cx + 28, cy + 8 - off); canvas_.print('z');
        canvas_.setCursor(cx + 38, cy + 2 - off); canvas_.print('Z');
    }
}
```
